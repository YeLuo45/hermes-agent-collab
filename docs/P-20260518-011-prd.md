# Direction J: 监控指标持久化（Prometheus + Grafana）

## 目标

为 hermes-agent-collab 添加 Prometheus 监控指标导出能力，提供标准 `/metrics` 端点，支持 gauges/counters/histograms，接入 Grafana 可视化。

## 背景

当前 `plugin_system.py` 有内存版的 `TaskMetricsPlugin` + `AgentMetricsPlugin`，但：
1. 指标只存在于 Python 进程内存，无法跨进程/跨重启持久化
2. 无标准 Exporter 端点（Prometheus 拉取需要 `/metrics`）
3. 无 Histogram 分布数据（只有累计计数）
4. 后续可接 PostgreSQL 时序数据做长期存储

Direction J 将内存指标提升为 Prometheus-compatible 的 Exporter，同时支持 PostgreSQL 时序存储。

### 核心类设计

```python
# collaboration/metrics.py

from prometheus_client import (
    Counter, Gauge, Histogram, CollectorRegistry, generate_latest,
    CONTENT_TYPE_LATEST, REGISTRY
)

class MetricsRegistry:
    """Singleton Prometheus metrics registry with helper methods."""

    _instance = None

    def __init__(self):
        self._counters: dict[str, Counter] = {}
        self._gauges: dict[str, Gauge] = {}
        self._histograms: dict[str, Histogram] = {}
        self._task_duration_buckets = (0.1, 0.5, 1, 5, 10, 30, 60, 300, float('inf'))
        self._api_duration_buckets = (0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1, 2.5, 5, 10)

    @classmethod
    def get_instance(cls) -> "MetricsRegistry":
        if cls._instance is None:
            cls._instance = cls()
        return cls._instance

    def counter(self, name: str, description: str, labels: list[str] = None) -> Counter:
        if name not in self._counters:
            self._counters[name] = Counter(name, description, labels or [])
        return self._counters[name]

    def gauge(self, name: str, description: str, labels: list[str] = None) -> Gauge:
        if name not in self._gauges:
            self._gauges[name] = Gauge(name, description, labels or [])
        return self._gauges[name]

    def histogram(self, name: str, description: str, buckets: tuple, labels: list[str] = None) -> Histogram:
        if name not in self._histograms:
            self._histograms[name] = Histogram(name, description, labels or [], buckets=buckets)
        return self._histograms[name]

    # ─── Pre-defined metrics ───────────────────────────────────────────────

    def init_task_metrics(self):
        c = self.counter("hermes_task_created_total", "Total tasks created", ["complexity"])
        self.counter("hermes_task_completed_total", "Total tasks completed", ["complexity"])
        self.counter("hermes_task_failed_total", "Total tasks failed", ["complexity"])
        self.histogram("hermes_task_duration_seconds", "Task duration", self._task_duration_buckets, ["complexity"])
        self.gauge("hermes_task_in_flight", "Tasks in flight", ["status"])
        self.gauge("hermes_task_queue_depth", "Task queue depth")

    def init_agent_metrics(self):
        self.counter("hermes_agent_registered_total", "Total agents registered", ["role"])
        self.counter("hermes_agent_status_changes_total", "Agent status changes", ["role", "status"])
        self.gauge("hermes_agent_active", "Active agents count")
        self.gauge("hermes_agent_idle", "Idle agents count")

    def init_hook_metrics(self):
        self.counter("hermes_hooks_emitted_total", "Total hook emissions", ["event"])
        self.counter("hermes_hooks_failed_total", "Failed hook emissions", ["event"])
        self.histogram("hermes_hook_handler_duration_seconds", "Hook handler duration",
                       (0.001, 0.005, 0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1, 2.5), ["event"])

    def init_api_metrics(self):
        self.counter("hermes_api_request_total", "API requests", ["method", "endpoint", "status"])
        self.histogram("hermes_api_request_duration_seconds", "API request duration",
                       self._api_duration_buckets, ["endpoint"])
        self.gauge("hermes_workspace_count", "Active workspaces count")

    def emit_metrics(self):
        """Initialize all metric groups."""
        self.init_task_metrics()
        self.init_agent_metrics()
        self.init_hook_metrics()
        self.init_api_metrics()
```

### MetricsPlugin 集成

```python
# collaboration/metrics_plugin.py

from collaboration.models import HookEvent, Plugin
from collaboration.metrics import MetricsRegistry

class MetricsPlugin(Plugin):
    """Plugin that bridges lifecycle hooks to Prometheus metrics."""

    def __init__(self):
        super().__init__(
            plugin_id="builtin:metrics",
            name="Prometheus Metrics",
            description="Exposes hook/metrics as Prometheus /metrics endpoint",
            version="1.0.0",
            enabled=True,
            hook_handlers={
                HookEvent.TASK_CREATED: ["_on_task_created"],
                HookEvent.TASK_COMPLETED: ["_on_task_completed"],
                HookEvent.TASK_FAILED: ["_on_task_failed"],
                HookEvent.AGENT_REGISTERED: ["_on_agent_registered"],
                HookEvent.AGENT_STATUS_CHANGED: ["_on_agent_status_changed"],
            },
        )
        self._metrics = MetricsRegistry.get_instance()
        self._metrics.emit_metrics()  # ensure all metrics registered

    def _on_task_created(self, payload):
        self._metrics.counter("hermes_task_created_total", ...).labels(
            complexity=payload.get("complexity", "normal")
        ).inc()

    def _on_task_completed(self, payload):
        self._metrics.counter("hermes_task_completed_total", ...).labels(
            complexity=payload.get("complexity", "normal")
        ).inc()
        if "duration" in payload:
            self._metrics.histogram("hermes_task_duration_seconds", ...).labels(
                complexity=payload.get("complexity", "normal")
            ).observe(payload["duration"])

    # ... similar for agent and hook metrics
```

### FastAPI `/metrics` 端点

```python
# collab_api.py

from prometheus_client import generate_latest, CONTENT_TYPE_LATEST
from collaboration.metrics import MetricsRegistry

@app.get("/metrics")
async def metrics():
    """Prometheus /metrics endpoint."""
    MetricsRegistry.get_instance().emit_metrics()  # ensure all registered
    return Response(content=generate_latest(), media_type=CONTENT_TYPE_LATEST)
```

### 可选：PostgreSQL 时序存储

当 `config.json` 中 `storage_backend: "postgres"` 时，MetricsPlugin 将指标样本同时写入 PostgreSQL `metrics` 表：

```sql
CREATE TABLE IF NOT EXISTS metrics (
    id SERIAL PRIMARY KEY,
    metric_name TEXT NOT NULL,
    labels JSONB DEFAULT '{}',
    value FLOAT,
    timestamp TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX IF NOT EXISTS idx_metrics_name_time ON metrics(metric_name, timestamp DESC);
```

```python
# In MetricsPlugin - async write to PostgreSQL on each metric update
async def _write_metric_async(self, name: str, labels: dict, value: float):
    if self._pg_dsn:
        pool = await asyncpg.create_pool(self._pg_dsn)
        async with pool.acquire() as conn:
            await conn.execute(
                "INSERT INTO metrics (metric_name, labels, value) VALUES ($1, $2, $3)",
                name, json.dumps(labels), value
            )
```
