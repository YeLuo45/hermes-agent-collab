# Direction K: Docker Compose 部署配置

## 目标

为 hermes-agent-collab 提供一键 Docker Compose 部署方案，支持 PostgreSQL + Redis + App 全家桶启动，提供开发/生产两套配置。

## 架构设计

### 新增文件

| 文件 | 说明 |
|------|------|
| `Dockerfile` | Python 3.12 slim, multi-stage build |
| `docker-compose.yml` | 开发配置：postgres + redis + app |
| `docker-compose.prod.yml` | 生产配置：postgres + redis + app + nginx |
| `.dockerignore` | 排除 node_modules/tests/.git |
| `docker/entrypoint.sh` | 容器启动脚本（DB 初始化 + 迁移） |
| `docker/nginx.conf` | 生产 Nginx 反向代理配置 |

### Dockerfile (multi-stage)

```dockerfile
# ── Builder stage ──────────────────────────────────────────
FROM python:3.12-slim AS builder

WORKDIR /app
RUN apt-get update && apt-get install -y --no-install-recommends \
    build-essential libpq-dev \
    && rm -rf /var/lib/apt/lists/*

COPY requirements.txt .
RUN pip install --no-cache-dir --user -r requirements.txt

# ── Runtime stage ──────────────────────────────────────────
FROM python:3.12-slim

WORKDIR /app

# Install runtime deps only
RUN apt-get update && apt-get install -y --no-install-recommends \
    libpq5 curl \
    && rm -rf /var/lib/apt/lists/* \
    && useradd --create-home hermes

COPY --from=builder /root/.local /home/hermes/.local
COPY --chown=hermes:hermes . .

# Hermes data volume
ENV HERMES_HOME=/home/hermes/.hermes
RUN mkdir -p $HERMES_HOME/workspaces

USER hermes
ENV PATH=/home/hermes/.local/bin:$PATH
ENV PYTHONUNBUFFERED=1

EXPOSE 8000

ENTRYPOINT ["python", "-m", "uvicorn"]
CMD ["collaboration.collab_api:app", "--host", "0.0.0.0", "--port", "8000"]
```

### docker-compose.yml（开发）

```yaml
version: "3.9"

services:
  app:
    build:
      context: .
      dockerfile: Dockerfile
    ports:
      - "8000:8000"
    environment:
      - HERMES_HOME=/home/hermes/.hermes
      - STORAGE_BACKEND=postgres
      - POSTGRES_HOST=postgres
      - POSTGRES_PORT=5432
      - POSTGRES_DB=hermes_collab
      - POSTGRES_USER=hermes
      - POSTGRES_PASSWORD=hermes_secret
      - REDIS_HOST=redis
      - REDIS_PORT=6379
    volumes:
      - hermes_data:/home/hermes/.hermes
      - ./dashboard:/app/dashboard:ro
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_started
    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:8000/monitor/health"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 20s
    restart: unless-stopped

  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: hermes_collab
      POSTGRES_USER: hermes
      POSTGRES_PASSWORD: herems_secret
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U hermes -d hermes_collab"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: unless-stopped

  redis:
    image: redis:7-alpine
    volumes:
      - redis_data:/data
    restart: unless-stopped

volumes:
  hermes_data:
  postgres_data:
  redis_data:
```

### docker-compose.prod.yml（生产）

```yaml
version: "3.9"

services:
  app:
    build:
      context: .
      dockerfile: Dockerfile
    environment:
      - HERMES_HOME=/home/hermes/.hermes
      - STORAGE_BACKEND=postgres
      - POSTGRES_HOST=postgres
      - POSTGRES_PORT=5432
      - POSTGRES_DB=hermes_collab
      - POSTGRES_USER=hermes
      - POSTGRES_PASSWORD=${POSTGRES_PASSWORD}
      - REDIS_HOST=redis
      - REDIS_PORT=6379
      - API_KEY_SECRET=${API_KEY_SECRET}
    volumes:
      - hermes_data:/home/hermes/.hermes
    depends_on:
      postgres:
        condition: service_healthy
      redis:
        condition: service_started
    restart: always
    deploy:
      resources:
        limits:
          memory: 1G
        reservations:
          memory: 256M

  postgres:
    image: postgres:16-alpine
    environment:
      POSTGRES_DB: hermes_collab
      POSTGRES_USER: hermes
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U hermes -d hermes_collab"]
      interval: 10s
      timeout: 5s
      retries: 5
    restart: always

  redis:
    image: redis:7-alpine
    command: redis-server --appendonly yes
    volumes:
      - redis_data:/data
    restart: always

  nginx:
    image: nginx:alpine
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./docker/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./dashboard:/usr/share/nginx/html:ro
    depends_on:
      - app
    restart: always

volumes:
  hermes_data:
  postgres_data:
  redis_data:
```

### entrypoint.sh

```bash
#!/bin/bash
set -e

echo "Waiting for PostgreSQL..."
until pg_isready -h $POSTGRES_HOST -p $POSTGRES_PORT -U $POSTGRES_USER; do
  sleep 2
done
echo "PostgreSQL ready."

echo "Creating database if not exists..."
PGPASSWORD=$POSTGRES_PASSWORD psql -h $POSTGRES_HOST -p $POSTGRES_PORT -U $POSTGRES_USER -d postgres -c "SELECT 1 FROM pg_database WHERE datname='$POSTGRES_DB'" | grep -q 1 || \
  PGPASSWORD=$POSTGRES_PASSWORD psql -h $POSTGRES_HOST -p $POSTGRES_PORT -U $POSTGRES_USER -d postgres -c "CREATE DATABASE $POSTGRES_DB"

echo "Ready. Starting hermes-agent-collab..."
exec "$@"
```

### Nginx 配置

```nginx
events {
    worker_connections 1024;
}

http {
    include /etc/nginx/mime.types;
    default_type application/octet-stream;

    upstream app {
        server app:8000;
        keepalive 32;
    }

    server {
        listen 80;
        server_name _;

        # Serve dashboard static files
        location /dashboard {
            alias /usr/share/nginx/html;
            try_files $uri $uri/ /index.html;
        }

        # Proxy API to FastAPI
        location /api/ {
            proxy_pass http://app;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_read_timeout 300s;
        }

        # Proxy metrics
        location /metrics {
            proxy_pass http://app;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
        }

        # Proxy SSE
        location /sse {
            proxy_pass http://app;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
            proxy_cache off;
            proxy_buffering off;
        }
    }
}
```
